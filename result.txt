struct Food<T> {
    id: u32,
    name: String,
    genre: u32,
    param: T,
}
struct FoodBuilder<T: ::core::default::Default> {
    id: Option<u32>,
    name: Option<String>,
    genre: Option<u32>,
    param: Option<T>,
}
impl<T: ::core::default::Default> FoodBuilder<T> {
    pub fn id(mut self, value: u32) -> Self {
        self.id = Some(value);
        self
    }
    pub fn name(mut self, value: String) -> Self {
        self.name = Some(value);
        self
    }
    pub fn genre(mut self, value: u32) -> Self {
        self.genre = Some(value);
        self
    }
    pub fn param(mut self, value: T) -> Self {
        self.param = Some(value);
        self
    }
    pub fn build(self) -> Food {
        Food {
            id: self.id.unwrap_or_default(),
            name: self.name.unwrap_or_default(),
            genre: self.genre.unwrap_or_default(),
            param: self.param.unwrap_or_default(),
        }
    }
}
impl<T: ::core::default::Default> Food<T> {
    pub fn builder() -> FoodBuilder {
        FoodBuilder {
            id: None,
            name: None,
            genre: None,
            param: None,
        }
    }
}
impl<T: std::fmt::Debug> std::fmt::Display for Food<T> {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        f.write_fmt(
            format_args!(
                "{0} {{ {1} }}",
                "Food",
                <[_]>::into_vec(
                        #[rustc_box]
                        ::alloc::boxed::Box::new([
                            ::alloc::__export::must_use({
                                let res = ::alloc::fmt::format(
                                    format_args!("{0}: {1:?}", "id", self.id),
                                );
                                res
                            }),
                            ::alloc::__export::must_use({
                                let res = ::alloc::fmt::format(
                                    format_args!("{0}: {1:?}", "name", self.name),
                                );
                                res
                            }),
                            ::alloc::__export::must_use({
                                let res = ::alloc::fmt::format(
                                    format_args!("{0}: {1:?}", "genre", self.genre),
                                );
                                res
                            }),
                            ::alloc::__export::must_use({
                                let res = ::alloc::fmt::format(
                                    format_args!("{0}: {1:?}", "param", self.param),
                                );
                                res
                            }),
                        ]),
                    )
                    .join(" "),
            ),
        )
    }
}
struct ItemPrice {
    id: u32,
    item_id: u32,
    end_date: Option<String>,
}
struct ItemPriceBuilder {
    id: Option<u32>,
    item_id: Option<u32>,
    end_date: Option<String>,
}
impl ItemPriceBuilder {
    pub fn id(mut self, value: u32) -> Self {
        self.id = Some(value);
        self
    }
    pub fn item_id(mut self, value: u32) -> Self {
        self.item_id = Some(value);
        self
    }
    pub fn end_date(mut self, value: Option<String>) -> Self {
        self.end_date = value;
        self
    }
    pub fn build(self) -> ItemPrice {
        ItemPrice {
            id: self.id.unwrap_or_default(),
            item_id: self.item_id.unwrap_or_default(),
            end_date: self.end_date,
        }
    }
}
impl ItemPrice {
    pub fn builder() -> ItemPriceBuilder {
        ItemPriceBuilder {
            id: None,
            item_id: None,
            end_date: None,
        }
    }
}
